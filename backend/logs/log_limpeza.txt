@echo off
:: 🔇 Oculta os comandos do terminal para deixar a saída limpa

chcp 65001 >nul
:: 🌐 Define UTF-8 para permitir acentos e emojis

:: 📁 Define o caminho do script Python
set "CAMINHO_PYTHON=C:\Users\Raquel\AppData\Local\Programs\Python\Python312\python.exe"
set "SCRIPT_PY=D:\Users\Raquel\Desktop\projeto_obemtequer\backend\limpar_imagens_antigas.py"

:: 📝 Define o caminho do arquivo de log (vai criar/atualizar este arquivo)
set "ARQUIVO_LOG=D:\Users\Raquel\Desktop\projeto_obemtequer\logs\log_limpeza.txt"

:: ⏰ Cria timestamp legível para registrar no log
for /f %%a in ('powershell -command "Get-Date -Format yyyy-MM-dd_HH-mm-ss"') do set "TIMESTAMP=%%a"

:: 🧹 Início
echo 🧹 Iniciando limpeza de imagens antigas... (%TIMESTAMP%) >> "%ARQUIVO_LOG%"
echo ------------------------------------------------------------ >> "%ARQUIVO_LOG%"

:: 🚀 Executa o script Python e redireciona saída (stdout + stderr) para o log
"%CAMINHO_PYTHON%" "%SCRIPT_PY%" >> "%ARQUIVO_LOG%" 2>&1

:: 📦 Verifica se o script foi executado com sucesso (código de saída = 0)
if %errorlevel%==0 (
    echo ✅ Limpeza concluída com sucesso em %TIMESTAMP%. >> "%ARQUIVO_LOG%"
    echo ✅ Limpeza concluída com sucesso!
) else (
    echo ❌ Ocorreu um erro na execução do script em %TIMESTAMP%. >> "%ARQUIVO_LOG%"
    echo ❌ Ocorreu um erro ao executar o script Python!
)

echo ------------------------------------------------------------ >> "%ARQUIVO_LOG%"
echo. >> "%ARQUIVO_LOG%"  :: Linha em branco no log para separar execuções

pause
:: ⏸️ Mantém o terminal aberto para o usuário visualizar a saída
